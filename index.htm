function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var cats = ["POPULASI_UNIT","MAINTENANCE","TROUBLESHOOT","DISMANTLE","INSTALLASI","REPLACE_CPE","STOK_CPE_READY","CPE_RUSAK"];
  var data = {};
  cats.forEach(function(name) {
    var sh = ss.getSheetByName(name);
    if (!sh || sh.getLastRow() < 2) { data[name] = []; return; }
    var vals = sh.getDataRange().getValues(), h = vals[0], arr = [];
    for (var i = 1; i < vals.length; i++) {
      var o = {};
      for (var j = 0; j < h.length; j++) {
        if (h[j] && h[j] !== "synced_at") o[h[j]] = String(vals[i][j]).replace(/^'/, "");
      }
      arr.push(o);
    }
    data[name] = arr;
  });
  return ContentService.createTextOutput(JSON.stringify({status:"SUCCESS", data:data})).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var body = JSON.parse(e.postData.contents);
    var cat = body.category, p = body.payload || {};
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = cat.replace(/_(EDIT|DELETE)$/, "");
    if (cat === "CLEAR_POPULASI_UNIT") {
      var shClear = ss.getSheetByName("POPULASI_UNIT");
      if (shClear && shClear.getLastRow() > 1) shClear.deleteRows(2, shClear.getLastRow() - 1);
      return ContentService.createTextOutput(JSON.stringify({status:"CLEARED"})).setMimeType(ContentService.MimeType.JSON);
    }
    var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
    var keys = Object.keys(p);
    if (sh.getLastRow() === 0 && keys.length > 0) sh.appendRow(keys.concat(["synced_at"]));
    var data = sh.getDataRange().getValues(), headers = data[0] || keys;
    var idCol = headers.indexOf("id"), rowIdx = -1;
    if (idCol !== -1 && p.id) {
      for (var r = 1; r < data.length; r++) { if (String(data[r][idCol]) === String(p.id)) { rowIdx = r + 1; break; } }
    }
    if (cat.endsWith("_DELETE") && rowIdx !== -1) { sh.deleteRow(rowIdx); }
    else if (!cat.endsWith("_DELETE")) {
      var rowVals = headers.map(function(h) { return h === "synced_at" ? new Date().toISOString() : (p[h] !== undefined ? "'" + p[h] : ""); });
      if (rowIdx !== -1) sh.getRange(rowIdx, 1, 1, rowVals.length).setValues([rowVals]);
      else sh.appendRow(rowVals);
    }
    return ContentService.createTextOutput(JSON.stringify({status:"OK"})).setMimeType(ContentService.MimeType.JSON);
  } finally { lock.releaseLock(); }
}
