function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var cats = ["DASHBOARD_SUMMARY","UNIT_CATEGORY_SUMMARY","POPULASI_UNIT","MAINTENANCE","TROUBLESHOOT","DISMANTLE","INSTALLASI","REPLACE_CPE","STOK_CPE_READY","CPE_RUSAK","USER_ACCOUNTS"];
  var data = {};
  cats.forEach(function(name) {
    var sh = ss.getSheetByName(name);
    if (!sh || sh.getLastRow() < 2) { data[name] = []; return; }
    var vals = sh.getDataRange().getValues(), h = vals[0], arr = [];
    for (var i = 1; i < vals.length; i++) {
      var o = {};
      for (var j = 0; j < h.length; j++) {
        if (h[j] && h[j] !== "synced_at") o[h[j]] = String(vals[i][j]).replace(/^'/, "");
      }
      arr.push(o);
    }
    data[name] = arr;
  });
  return ContentService.createTextOutput(JSON.stringify({status:"SUCCESS", data:data})).setMimeType(ContentService.MimeType.JSON);
}
function upsertSheetRows(ss, sheetName, rows, replaceAll) {
  var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  if (replaceAll) {
    sh.clearContents();
    if (!rows || rows.length === 0) return;
    var keySet = {};
    rows.forEach(function(r) {
      Object.keys(r || {}).forEach(function(k) { if (k && k !== "synced_at") keySet[k] = true; });
    });
    var keys = Object.keys(keySet);
    var headers = keys.concat(["synced_at"]);
    var nowIso = new Date().toISOString();
    var out = [headers];
    rows.forEach(function(r) {
      out.push(headers.map(function(h) { return h === "synced_at" ? nowIso : (r[h] !== undefined && r[h] !== null ? "'" + r[h] : ""); }));
    });
    sh.getRange(1, 1, out.length, headers.length).setValues(out);
  }
}
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var body = JSON.parse(rawBody);
    var cat = body.category || "", p = body.payload || {};
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (body.summary && body.summary.DASHBOARD_SUMMARY) {
      upsertSheetRows(ss, "DASHBOARD_SUMMARY", body.summary.DASHBOARD_SUMMARY, true);
    }
    if (body.summary && body.summary.UNIT_CATEGORY_SUMMARY) {
      upsertSheetRows(ss, "UNIT_CATEGORY_SUMMARY", body.summary.UNIT_CATEGORY_SUMMARY, true);
    }
    if (cat === "FULL_STATE_SYNC" && body.state) {
      Object.keys(body.state).forEach(function(sheetName) {
        upsertSheetRows(ss, sheetName, body.state[sheetName] || [], true);
      });
      return ContentService.createTextOutput(JSON.stringify({status:"FULL_SYNCED"})).setMimeType(ContentService.MimeType.JSON);
    }
    var sheetName = cat.replace(/_(EDIT|DELETE)$/, "");
    if (!sheetName) return ContentService.createTextOutput(JSON.stringify({status:"IGNORED"})).setMimeType(ContentService.MimeType.JSON);
    var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
    var pKeys = Object.keys(p).filter(function(k) { return k && k !== "synced_at"; });
    if (sh.getLastRow() === 0 && pKeys.length > 0) sh.appendRow(pKeys.concat(["synced_at"]));
    var data = sh.getDataRange().getValues(), headers = (data[0] || []).map(String);
    pKeys.forEach(function(k) {
      if (headers.indexOf(k) === -1) {
        var insertCol = headers.indexOf("synced_at") !== -1 ? headers.indexOf("synced_at") + 1 : headers.length + 1;
        sh.insertColumnBefore(insertCol);
        sh.getRange(1, insertCol).setValue(k);
        headers.splice(insertCol - 1, 0, k);
      }
    });
    if (headers.indexOf("synced_at") === -1) {
      sh.getRange(1, headers.length + 1).setValue("synced_at");
      headers.push("synced_at");
    }
    var idCol = headers.indexOf("id"), rowIdx = -1;
    if (idCol !== -1 && p.id) {
      for (var r = 1; r < data.length; r++) { if (String(data[r][idCol]).replace(/^'/, "") === String(p.id)) { rowIdx = r + 1; break; } }
    }
    if (cat.endsWith("_DELETE")) {
      if (rowIdx !== -1) sh.deleteRow(rowIdx);
    } else {
      var nowIso = new Date().toISOString();
      var rowVals = headers.map(function(h) { return h === "synced_at" ? nowIso : (p[h] !== undefined && p[h] !== null ? "'" + p[h] : ""); });
      if (rowIdx !== -1) sh.getRange(rowIdx, 1, 1, rowVals.length).setValues([rowVals]);
      else sh.appendRow(rowVals);
    }
    return ContentService.createTextOutput(JSON.stringify({status:"OK"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status:"ERROR", message:String(err)})).setMimeType(ContentService.MimeType.JSON);
  } finally { lock.releaseLock(); }
}
